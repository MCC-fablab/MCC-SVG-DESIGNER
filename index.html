#include #include const char* ssid = "GhostCam01"; const char* password = "ghostcam01"; // スイッチの片側をGPIO 27、もう片側をGNDへ接続します。 // GPIO 27がすでに別の機器に使われている場合だけ、未使用のGPIOに変更してください。 constexpr uint8_t SWITCH_PIN = 27; constexpr unsigned long DEBOUNCE_MS = 40; WebServer server(80); bool ghostOn = false; bool lastReading = HIGH; bool stableReading = HIGH; unsigned long lastChangeAt = 0; void setGhost(bool on) { if (ghostOn == on) return; ghostOn = on; Serial.println(ghostOn ? "GHOST ON" : "GHOST OFF"); } void sendStatus() { server.send(200, "text/plain; charset=utf-8", ghostOn ? "GHOST_ON" : "GHOST_OFF"); } void setup() { Serial.begin(115200); delay(1000); pinMode(SWITCH_PIN, INPUT_PULLUP); lastReading = stableReading = digitalRead(SWITCH_PIN); ghostOn = (stableReading == LOW); WiFi.mode(WIFI_AP); if (!WiFi.softAP(ssid, password)) { Serial.println("AP start failed"); return; } server.on("/status", HTTP_GET, sendStatus); server.on("/on", HTTP_GET, []() { setGhost(true); sendStatus(); }); server.on("/off", HTTP_GET, []() { setGhost(false); sendStatus(); }); server.on("/", HTTP_GET, []() { server.send(200, "text/plain", "GhostCam is ready"); }); server.onNotFound([]() { server.send(404, "text/plain", "Not found: " + server.uri()); }); server.begin(); Serial.println("=== Ghost Camera AP ==="); Serial.print("SSID: "); Serial.println(ssid); Serial.print("IP address: "); Serial.println(WiFi.softAPIP()); Serial.println(ghostOn ? "GHOST ON" : "GHOST OFF"); } void loop() { server.handleClient(); const bool reading = digitalRead(SWITCH_PIN); if (reading != lastReading) lastChangeAt = millis(); if (millis() - lastChangeAt >= DEBOUNCE_MS && reading != stableReading) { stableReading = reading; setGhost(stableReading == LOW); // GNDにつながったときON } lastReading = reading; }
